% Chapter 2, Problem 5(e): field of an infinite homogeneous half-plane
% The half-plane is z = 0, x < 0 and its edge is the y axis. The field is the
% same in every cross-section y = const.: it is unchanged by translations along
% y only.

\begin{document}
\begin{tikzpicture}
    % Half-plane and its symmetry
    \begin{scope}[default view3d]
        \draw[axis] (0,0,0) -- (0,2.6,0) node[right] {$y$};
        \pic {halfplane3d={1.8}{1.9}};
        \draw[axis] (0,0,0) -- (1.3,0,0) node[below left=-2pt] {$x$};
        \draw[axis] (0,0,0) -- (0,0,1.8) node[above] {$z$};
        \draw[translation] (0.6,-1.5,0) -- node[symmetry label, below left=-2pt] {$P_y$} (0.6,-0.2,0);
    \end{scope}

    % Field in a cross-section
    \begin{fieldsection}[xshift=5cm]{-2.2}{1.8}{-1.6}{1.6}{cross-section $y = \text{const.}$}

        \FieldLinesSegment{-2.2}{1.7}{1.8}{1.7}{9}

        \draw[axis] (0,0) -- (1.2,0) node[axis label, below] {$x$};
        \draw[axis] (0,0) -- (0,1.2) node[axis label, right] {$z$};
    \end{fieldsection}
