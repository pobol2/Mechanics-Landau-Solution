% The field is the same in every cross-section perpendicular to the edges, but
% it is not symmetric about any axis: it is unchanged by translations along z only.

        \pic at (0,0,1.6) {ztranslation={1.2}{$P_z$}};

    % Field in a cross-section (edges at the azimuths 30, 150 and 270 degrees)
