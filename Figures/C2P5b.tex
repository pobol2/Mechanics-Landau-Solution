% Chapter 2, Problem 5(b): field of an infinite homogeneous cylinder
% The field is radial and depends only on the distance to the axis: it is
% unchanged by translations along z and by rotations about z.
\documentclass{SolutionFigure}

\begin{document}
\begin{tikzpicture}
    \def\R{0.7}     % radius of the cylinder

    % Cylinder and its symmetries
    \begin{scope}[default view3d]
        \pic {cylinder3d={\R}{1.3}};
        \draw[axis] (0,0,1.3) -- (0,0,3.3) node[above] {$z$};
        \pic at (0,0,1.6) {ztranslation={1.2}{$P_z$}};
        \pic at (0,0,2.3) {zrotation={0.4}{$M_z$}};
    \end{scope}

    % Field in a cross-section
    \begin{fieldsection}[xshift=5cm, yshift=0.3cm]{-2}{2}{-2}{2}{cross-section $z = \text{const.}$}
        \FieldDisc{0}{0}{\R}
        \Equipotentials{1.05/0, 1.5/0, 2.1/0, 2.7/0}
        \FieldLinesCircle{0}{0}{2.9}{16}{11.25}
        \filldraw[fill=body] (0,0) circle[radius=\R];
        \draw[axis] (0,0) -- (1.4,0) node[axis label, below] {$x$};
        \draw[axis] (0,0) -- (0,1.4) node[axis label, right] {$y$};
    \end{fieldsection}
\end{tikzpicture}
\end{document}
