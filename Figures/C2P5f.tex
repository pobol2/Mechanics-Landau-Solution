% Chapter 2, Problem 5(f): field of a homogeneous cone
% The field is symmetric about the axis of the cone: it is unchanged by
% rotations about it only.
\documentclass{SolutionFigure}

\begin{document}
\begin{tikzpicture}
    \def\R{1}       % radius of the base
    \def\H{1.9}     % height

    % Cone and its symmetry
    \begin{scope}[default view3d]
        \pic {cone3d={\R}{\H}};
        \draw[axis] (0,0,\H) -- (0,0,\H+1.6) node[above] {$z$};
        \pic at (0,0,\H+0.8) {zrotation={0.4}{$M_z$}};
    \end{scope}

    % Field in a plane containing the axis
    \begin{fieldsection}[xshift=4.5cm]{-2}{2}{-1.3}{2.7}{plane $y = 0$}
        \FieldCone{\R}{\H}
        \Equipotentials{1.15/0.3, 1.6/0.3, 2.3/0.3}
        \FieldLinesCircle{0}{0.5}{3.2}{22}{8}
        \filldraw[fill=body] (-\R,0) -- (\R,0) -- (0,\H) -- cycle;
        \draw[axis] (0,0) -- (1.5,0) node[axis label, below] {$x$};
        \draw[axis] (0,\H) -- (0,2.5) node[axis label, right] {$z$};
    \end{fieldsection}
\end{tikzpicture}
\end{document}
