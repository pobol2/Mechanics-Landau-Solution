% rotations about it only.

    \def\d{0.8}     % distance from the origin to each point

    % Points and their symmetry
    \begin{scope}[default view3d]

        \node[mass, minimum size=2mm] at (0,0,-\d) {};
        \pic at (0,0,2) {zrotation={0.4}{$M_z$}};

    % Field in a plane containing the points
    \begin{fieldsection}[xshift=4cm, yshift=0.5cm]{-2}{2}{-2}{2}{plane $y = 0$}

        \FieldLinesCircle{0}{0}{2.9}{20}{9}
        \fill (0,\d) circle[radius=0.1];

        \draw[axis] (0,0) -- (1.4,0) node[axis label, below] {$x$};
        \draw[axis] (0,\d) -- (0,1.8) node[axis label, right] {$z$};
    \end{fieldsection}
