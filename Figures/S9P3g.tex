% §9 Problem 3(g): field of a homogeneous circular torus
% The field is symmetric about the axis of the torus: it is unchanged by
% rotations about it only.
\documentclass{SolutionFigure}

\begin{document}
\begin{tikzpicture}
    \def\R{1.3}     % radius of the central circle
    \def\r{0.25}    % radius of the tube

    % Torus and its symmetry
    \begin{scope}[default view3d]
        \pic {torus3d={\R}{\r}};
        \draw[axis] (0,0,0) -- (0,0,2.3) node[above] {$z$};
        \pic at (0,0,1.5) {zrotation={0.4}{$M_z$}};
    \end{scope}

    % Field in a plane containing the axis
    \begin{fieldsection}[xshift=5cm]{-2.4}{2.4}{-1.7}{1.7}{plane $y = 0$}
        \FieldTorus{\R}{\r}
        \Equipotentials{1.7/0, 2.2/0, 0/0.6, 0/1.3}
        \FieldLinesCircle{0}{0}{3.2}{24}{7.5}
        \filldraw[fill=body] (-\R,0) circle[radius=\r] (\R,0) circle[radius=\r];
        \draw[axis] (0,0) -- (0.9,0) node[axis label, below] {$x$};
        \draw[axis] (0,0) -- (0,1.2) node[axis label, right] {$z$};
    \end{fieldsection}
\end{tikzpicture}
\end{document}
