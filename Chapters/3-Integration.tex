% ============ Problem 1 ============ %

\begin{problem}
{
Determine the period of oscillations of a simple pendulum (a particle of mass $m$ suspended by a string of length $l$ in a gravitational field) as a function of the amplitude of the oscillations.
}
{
The potential energy of a simple pendulum is
\begin{equation*}
   U(\phi) = - mgl\cos{\phi},
\end{equation*}
see \probref{\S 5}{1}. If we define $\phi_0$ as the maximum value of $\phi$, the potential energy is equal to the total energy at this point, that is
\begin{equation*}
    U(\phi_0) = E = -mgl\cos{\phi_0}.
\end{equation*}
The energy of the pendulum could also be write using the kinetic energy,
\begin{equation*}
    E = \frac{1}{2}ml^2\Dot{\phi}^2 + U(\phi).
\end{equation*}
This first order differential equation can be integrate to give the period of oscillation 
\begin{align*}
    T &= 2\sqrt{\frac{ml^2}{2}} \int_{-\phi_0}^{\phi_0} \frac{d\phi}{\sqrt{E-U(\phi)}} \\
    &= 4\sqrt{\frac{ml^2}{2}} \int_{0}^{\phi_0} \frac{d\phi}{\sqrt{mgl\cos{\phi}-mgl\cos{\phi_0}}} \\
    &= 4\sqrt{\frac{l}{2g}} \int_{0}^{\phi_0} \frac{d\phi}{\sqrt{\cos{\phi}-\cos{\phi_0}}}.
\end{align*}
To solve this integral, we need first to use a trigonometric identity (I'll let you find or derive this identity), giving us
\begin{equation*}
    T = 2\sqrt{\frac{l}{g}} \int_{0}^{\phi_0} \frac{d\phi}{\sqrt{\sin^2{\frac{\phi_0}{2}}-\sin^2{\frac{\phi}{2}}}}.
\end{equation*}
Next, we use the substitution
\begin{equation*}
    \sin{\xi} = \frac{\sin{\frac{\phi}{2}}}{\sin{\frac{\phi_0}{2}}}.
\end{equation*}
Its differential form is
\begin{align*}
    \frac{d\phi}{d\xi} &= \frac{d}{d\xi} \left( 2 \arcsin{\left(\sin{\frac{\phi_0}{2}}\sin{\xi}\right)} \right) \\
    &= \frac{2\sin{\frac{\phi_0}{2}}\cos{\xi}}{\sqrt{1-\sin^2{\frac{\phi_0}{2}}\sin^2{\xi}}}
\end{align*}
The substitution, finally, give
\begin{align*}
    T &= 2\sqrt{\frac{l}{g}} \int_{0}^{\phi_0} \frac{d\phi}{\sqrt{\sin^2{\frac{\phi_0}{2}} - \sin^2{\xi}\sin^2{\frac{\phi_0}{2}}}} \\
    &= 2\sqrt{\frac{l}{g}} \int_{0}^{\phi_0} \frac{d\phi}{\sin{\frac{\phi_0}{2}}\sqrt{ 1 - \sin^2{\xi}}} \\
    &= 2\sqrt{\frac{l}{g}} \int_{0}^{\phi_0} \frac{d\phi}{\sin{\frac{\phi_0}{2}}\cos{\xi}} \\
    &= 2\sqrt{\frac{l}{g}} \int_{\arcsin{(0)}}^{\arcsin{(1)}} \frac{2\sin{\frac{\phi_0}{2}}\cos{\xi} d\xi}{\sin{\frac{\phi_0}{2}}\cos{\xi}\sqrt{1-\sin^2{\frac{\phi_0}{2}}\sin^2{\xi}}} \\
    &= 4\sqrt{\frac{l}{g}} \int_{0}^{\frac{\pi}{2}} \frac{d\xi}{\sqrt{1-\sin^2{\frac{\phi_0}{2}}\sin^2{\xi}}}.
\end{align*}
We can also write
\begin{equation*}
    T = 4\sqrt{\frac{l}{g}} K\left(\sin{\frac{\phi_0}{2}}\right),
\end{equation*}
where 
\begin{equation} \label{S11P1_K}
    K(k) = \int_{0}^{\frac{\pi}{2}} \frac{d\xi}{\sqrt{1-k^2\sin^2{\xi}}}.
\end{equation}
The integral $K$ is known as the complete elliptic integral of the first kind. To solve this integral, we first need to see that the integrand is of the form 
\begin{equation*}
    f(x) = \left( 1 - x \right)^{-\frac{1}{2}}.
\end{equation*}
The $n$ derivative of this function is
\begin{equation*}
    f^{(n)}(x) = \frac{\left( 2n-1 \right)!!}{2^n}\left( 1-x \right)^{-\frac{2n+1}{2}},
\end{equation*}
thus the Maclaurin Series is
\begin{equation*}
    f(x) = \sum_{n\geq0}\frac{(2n-1)!!}{2^n \cdot n!}x^n.
\end{equation*}
Using this result in \eqref{S11P1_K}, we get
\begin{align*}
    K(k) &= \int_{0}^{\frac{\pi}{2}} \sum_{n\geq0}\frac{(2n-1)!!}{2^n \cdot n!}k^{2n}\sin^{2n}{\xi}\;d\xi \\
    &= \sum_{n\geq0}\frac{(2n-1)!!}{2^n \cdot n!}k^{2n} \int_{0}^{\frac{\pi}{2}} \sin^{2n}{\xi}\;d\xi.
\end{align*}
This last integral can be solve using the beta function
\begin{equation*}
    \mathcal{B}(x,y) = \frac{\Gamma(x)\Gamma(y)}{\Gamma(x+y)} = 2\int_0^{\frac{\pi}{2}} \sin^{2x-1}{t}\cos^{2y-1}{t}\;dt,
\end{equation*}
with $x=\frac{2n+1}{2}$ and $y=\frac{1}{2}$. Thus, using $\Gamma\left(n+\frac{1}{2}\right) = \frac{(2n-1)!!}{2^n}\sqrt{\pi}$, the beta function become
\begin{align*}
    \frac{1}{2}\mathcal{B}\left(n+\frac{1}{2},\frac{1}{2}\right) &= \frac{\Gamma(n+\frac{1}{2})\Gamma(\frac{1}{2})}{2\Gamma(n+1)} \\
    &= \frac{\sqrt{\pi}}{2}\frac{\Gamma(\frac{2n+1}{2})}{n!} \\
    &= \frac{\pi}{2}\frac{(2n-1)!!}{2^n \cdot n!}
\end{align*}
The elliptic integral is thereby
\begin{align*}
    K(k) &= \frac{\pi}{2} \sum_{n\geq0}\left[\frac{(2n-1)!!}{2^n \cdot n!}k^{n}\right]^2 \\
    &= \frac{\pi}{2} \sum_{n\geq0}\left[\frac{(2n-1)!!}{(2n)!!}k^{n}\right]^2.
\end{align*}
The period of a simple pendulum is
\begin{equation*}
    T = 2\pi\sqrt{\frac{l}{g}}\sum_{n\geq0}\left[\frac{(2n-1)!!}{(2n)!!}\sin^n{\frac{\phi_0}{2}}\right]^2
\end{equation*}
By expanding the sum, we get
\begin{equation*}
    T = 2\pi\sqrt{\frac{l}{g}}\left( 1 + \left(\frac{1}{2}\right)^2\sin^2{\frac{\phi_0}{2}} + \left(\frac{1\cdot3}{2\cdot4}\right)^2\sin^4{\frac{\phi_0}{2}} + \left(\frac{1\cdot3\cdot5}{2\cdot4\cdot6}\right)^2\sin^6{\frac{\phi_0}{2}} + \ldots \right).
\end{equation*}
Using the Maclaurin series
\begin{equation*}
    \sin{\frac{\phi_0}{2}} = \frac{1}{2}\phi_0 - \frac{1}{48}\phi_0^3 + \frac{1}{3840}\phi_0^5 - \frac{1}{645120}\phi_0^7 + \ldots,
\end{equation*}
we finally get
}

\begin{equation*}
    T = 2\pi\sqrt{\frac{l}{g}}\left( 1 + \frac{1}{16}\phi_0^2 + \frac{11}{3072}\phi_0^4 + \ldots \right)
\end{equation*}
}

For the motion to be an oscillation, we assume $A>0$, $n>0$ and $E>0$. The total energy of the particle is
\begin{equation*}
    E = \frac{1}{2}m\Dot{x}^2 + U(x).
\end{equation*}
At the turning points $x_1$ of the motion, the kinetic energy vanishes, so they satisfy $E = U(x_1)$, that is
\begin{align*}
    &E = U(x_1) = A|x_1|^n \\
    \Rightarrow &|x_1| = \left( \frac{E}{A} \right)^{1/n}, \\
     \Rightarrow &x_1 = \pm \left( \frac{E}{A} \right)^{1/n},
\end{align*}
we get the period of oscillation,
\begin{align*}
    T &= 2\sqrt{\frac{m}{2}} \int_{-\left( \frac{E}{A} \right)^{1/n}}^{\left( \frac{E}{A}\right)^{1/n} }\frac{dx}{\sqrt{E-A|x|^n}} \\
    &= 2\sqrt{2m} \int_0^{\left( \frac{E}{A}\right)^{1/n} }\frac{dx}{\sqrt{E-A|x|^n}} .
\end{align*}
It is possible to remove the absolute value, because the integral is over positive $x$, thus
\begin{align*}
    T &= 2\sqrt{2m} \int_0^{\left( \frac{E}{A}\right)^{1/n} }\frac{dx}{\sqrt{E-Ax^n}} \\
    &= 2\sqrt{\frac{2m}{E}} \int_0^{\left( \frac{E}{A}\right)^{1/n} }\frac{dx}{\sqrt{1-\frac{A}{E}x^n}}.
\end{align*}

\begin{align*}

\end{align*}

\begin{problem}
{
A system consists of one particle of mass $M$ and $n$ particles with equal masses $m$. Eliminate the motion of the centre of mass and so reduce the problem to one involving $n$ particles.
}
{
Let $\mathbf{R}$ be the radius vector of the particle of mass $M$, and $\mathbf{R}_a \left( a=1, 2, \ldots, n \right)$ those of the particles of mass $m$. We put $\mathbf{r}_a \equiv \mathbf{R}_a - \mathbf{R}$ and take the origin to be at the centre of mass, namely
\begin{equation*}
    M\mathbf{R} + m\sum_a \mathbf{R}_a = 0.
\end{equation*}
Thus, we got
\begin{equation} \label{S13P1_R}
    \mathbf{R} = -\frac{m}{M}\left(\sum_a\mathbf{r}_a + n\mathbf{R}\right) = -\frac{m}{\mu} \sum_a \mathbf{r}_a,
\end{equation}
where $\mu \equiv M + nm$. The Lagrangian of the system is
\begin{equation*}
    L = \frac{1}{2}M\Dot{\mathbf{R}}^2 + \frac{1}{2}m\sum_a\Dot{\mathbf{R}}_a^2 - U.
\end{equation*}
If we substitute \eqref{S13P1_R} and $\mathbf{R}_a = \mathbf{R} + \mathbf{r}_a$, we get
\begin{align*}
    L &= \frac{1}{2}\frac{Mm^2}{\mu^2} \left( \sum_a \Dot{\mathbf{r}}_a \right)^2 + \frac{1}{2}nm\Dot{\mathbf{R}}^2 + \frac{1}{2}m\sum_a\Dot{\mathbf{r}}_a^2 + m\Dot{\mathbf{R}}\sum_a \Dot{\mathbf{r}}_a - U \\
    &= \frac{1}{2}\frac{Mm^2}{\mu^2} \left( \sum_a \Dot{\mathbf{r}}_a \right)^2 + \frac{1}{2}\frac{nm^3}{\mu^2}\left( \sum_a \Dot{\mathbf{r}}_a \right)^2 + \frac{1}{2}m\sum_a\Dot{\mathbf{r}}_a^2 - \frac{m^2}{\mu}\left(\sum_a \Dot{\mathbf{r}}_a\right)^2 - U \\
    &= \frac{1}{2}m\sum_a\Dot{\mathbf{r}}_a^2 - \frac{1}{2}\frac{m^2}{\mu}\left( \frac{2\mu-M-nm}{\mu} \right)\left(\sum_a \Dot{\mathbf{r}}_a\right)^2 - U \\
    &= \frac{1}{2}m\sum_a\Dot{\mathbf{r}}_a^2 - \frac{1}{2}\frac{m^2}{\mu}\left(\sum_a \Dot{\mathbf{r}}_a\right)^2 - U.
\end{align*}
The potential energy depends only on the relative positions of the particles, so $U = U(\mathbf{r}_1, \ldots, \mathbf{r}_n)$. The Lagrangian is thus only a function of $\mathbf{r}_a$ and their time derivatives, which is
}
{
\begin{equation*}
    L = \frac{1}{2}m\sum_a\Dot{\mathbf{r}}_a^2 - \frac{1}{2}\frac{m^2}{\mu}\left(\sum_a \Dot{\mathbf{r}}_a\right)^2 - U(\mathbf{r}_1, \ldots, \mathbf{r}_n), \qquad \mu = M + nm
\end{equation*}
}
\end{problem}

% ============ Problem 1 ============ %

\begin{problem}
{
Integrate the equations of motion for a spherical pendulum (a particle of mass $m$ moving on the surface of a sphere of radius $l$ in a gravitational field).
\begin{figure}[H]
    \centering
    \includegraphics{Figures/S14P1.pdf}
\end{figure}
}
{
The Cartesian position of the particle $m$ is
\begin{align*}
    x &= l\sin{\theta}\cos{\phi} \\
    y &= l\sin{\theta}\sin{\phi} \\
    z &= -l\cos{\theta}.
\end{align*}
The time derivatives of these coordinates are
\begin{align*}
    \Dot{x} &= l\Dot{\theta}\cos{\theta}\cos{\phi} - l\Dot{\phi}\sin{\theta}\sin{\phi} \\
    \Dot{y} &= l\Dot{\theta}\cos{\theta}\sin{\phi} + l\Dot{\phi}\sin{\theta}\cos{\phi} \\
    \Dot{z} &= l\Dot{\theta}\sin{\theta}.
\end{align*}
Substituting these expressions in the Lagrangian, we obtain the desired Lagrangian form, that is
\begin{align*}
    L &= \frac{1}{2}m\left(\Dot{x}^2+\Dot{y}^2+\Dot{z}^2\right) - mgz \\
    &= \frac{1}{2}ml^2\left(\Dot{\theta}^2+\Dot{\phi}^2\sin^2{\theta}\right) + mgl\cos{\theta}.
\end{align*}
The Lagrangian does not involve $\phi$ explicitly, thus this co-ordinate is cyclic and the generalized momentum $p_\phi$ is conserved. This momentum is the same as the $z$-component of angular momentum $M_z$ (see \probref{\S 9}{2} with $r = l$) and is written
\begin{equation} \label{S14P1_Mz}
    M_z = p_{\phi} = \pd{L}{\Dot{\phi}} = ml^2\Dot{\phi}\sin^2{\theta}.
\end{equation}
The energy of the pendulum is
\begin{equation*}
    E = \frac{1}{2}ml^2\left(\Dot{\theta}^2+\Dot{\phi}^2\sin^2{\theta}\right) - mgl\cos{\theta}.
\end{equation*}
Substituting $\Dot{\phi}$ with \eqref{S14P1_Mz} in the energy we get
\begin{align*}
    E &= \frac{1}{2}ml^2\Dot{\theta}^2 + \frac{M_z^2}{2ml^2\sin^2{\theta}} - mgl\cos{\theta} \\
    &= \frac{1}{2}ml^2\Dot{\theta}^2 + U_{\text{eff}}(\theta),
\end{align*}
where $U_{\text{eff}}(\theta)$ is the effective potential energy. Notice that the energy was re-written as a function of only one co-ordinate (\ie $\theta$). This is equivalent to the one-dimensional problem we have already seen. On a branch of the motion where $\theta$ increases, $\Dot{\theta}$ is the positive square root, thus
\begin{align*}
    & \frac{d\theta}{dt} = \sqrt{\frac{2\left( E - U_{\text{eff}}(\theta) \right)}{ml^2}} \\
    \Rightarrow & t - t_0 = l\sqrt{\frac{m}{2}} \int_{\theta_0}^{\theta} \frac{d\vartheta}{\sqrt{E - U_{\text{eff}}(\vartheta)}} ,
\end{align*}
where $\theta_0 = \theta(t_0)$ and $\vartheta$ is the integration variable. On a branch where $\theta$ decreases, $\Dot{\theta}$ is the negative square root, and the right-hand side changes sign. This integral lead to an elliptic integral of the first kind see \probref{\S 11}{1}. We also need to find the solution to the angle $\phi$. To do so, we can use the \eqref{S14P1_Mz} with the chain rule from calculus, that is, on an increasing-$\theta$ branch,
\begin{align*}
    \frac{M_z}{ml^2\sin^2{\theta}} &= \frac{d\phi}{dt} \\
    &= \frac{d\phi}{d\theta}\frac{d\theta}{dt} \\
    &= \frac{d\phi}{d\theta}\sqrt{\frac{2\left( E - U_{\text{eff}}(\theta) \right)}{ml^2}}.
\end{align*}
Finally, we get
\begin{align*}
    & \frac{d\phi}{d\theta} =  \frac{M_z}{l\sin^2{\theta}} \sqrt{\frac{1}{2m\left( E - U_{\text{eff}}(\theta) \right)}} \\
    \Rightarrow & \phi - \phi_0 = \frac{M_z}{l\sqrt{2m}}\int_{\theta_0}^{\theta}\frac{d\vartheta}{\sin^2{\vartheta}\sqrt{E - U_{\text{eff}}(\vartheta)}},
\end{align*}
with $\phi_0 = \phi(t_0)$. Again, the sign of the right-hand side is reversed on a decreasing-$\theta$ branch, so that $\phi$ always changes in the direction given by the sign of $M_z$, as \eqref{S14P1_Mz} requires. This integral lead to an elliptic integral of the third kind (I really don't want to solve this). When $\theta$ reaches a turning point, $\Dot{\theta}$ changes sign: the motion is continued by restarting both integrals from that turning point with the opposite sign, so that $t$ keeps increasing and $\phi$ keeps changing monotonically. It is possible to find the range of angle $\theta$. At the maximum and minimum value of $\theta$, $\Dot{\theta}=0$, while the azimuthal kinetic energy $M_z^2/(2ml^2\sin^2{\theta})$ can remain nonzero, thus
\begin{align*}
    E &= U_{\text{eff}} \\
    &= \frac{M_z^2}{2ml^2\sin^2{\theta}} - mgl\cos{\theta} \\
    &= \frac{M_z^2}{2ml^2\left(1-\cos^2{\theta}\right)} - mgl\cos{\theta}
\end{align*}
We can find a cubic algebraic equation for $\cos{\theta}$, that is
\begin{align*}
    0 &= \left( E + mgl\cos{\theta} \right) \left( 2ml^2 \left( 1 - \cos^2{\theta} \right) \right) - M_z^2 \\
      &= mgl\left(\cos^3{\theta} - \cos{\theta}\right) + E\left(\cos^2{\theta} - 1\right) + \frac{M_z^2}{2ml^2} .
\end{align*}
The centrifugal part of the effective potential $U_{\text{eff}}$, namely
\begin{equation*}
    \frac{M_z^2}{2ml^2\sin^2{\theta}},
\end{equation*}
is never negative. If $M_z \neq 0$, it is positive and, as we can see, diverges at $\theta \rightarrow 0$ and $\theta \rightarrow \pi$, thus the angle $\theta$ is bound away from the poles. This can be write as
\begin{align*}
  \theta &\in \left[ \theta_p, \theta_a\right] & 0 &< \theta_p \leq \theta_a < \pi \qquad (M_z \neq 0),
\end{align*}
where the subscripts of the angles $\theta_p$ and $\theta_a$ are for the perigee and apogee. This mean that the two roots of the cubic algebraic equation for $\cos{\theta}$ giving the turning points are bound between $-1$ and $+1$ (the cubic is positive at $\cos{\theta}=\pm 1$, so its third root is smaller than $-1$). Except for the solution with constant $\theta$ at the minimum of $U_{\text{eff}}$ (when $E$ equals this minimum, the two roots merge and $\theta_p = \theta_a$), the turning points are distinct, $\theta_p < \theta_a$, and $\theta$ oscillates between them. If $M_z = 0$, there is no centrifugal barrier: the motion is planar (a pendulum swinging, or revolving, in a fixed vertical plane) and it can reach a pole. To summarize, the solution of the equation of motion of the spherical pendulum is
}
{
\begin{align*}
  t - t_0 &= l\sqrt{\frac{m}{2}}\int_{\theta_0}^{\theta} \frac{d\vartheta}{\sqrt{E - U_{\text{eff}}(\vartheta)}} \\
  \phi - \phi_0 &= \frac{M_z}{l\sqrt{2m}}\int_{\theta_0}^{\theta}\frac{d\vartheta}{\sin^2{\vartheta}\sqrt{E - U_{\text{eff}}(\vartheta)}} \\
  U_{\text{eff}}(\theta) &= \frac{M_z^2}{2ml^2\sin^2{\theta}} - mgl\cos{\theta}
\end{align*}
on a branch where $\theta$ increases; both right-hand sides change sign on a branch where $\theta$ decreases, and the integrals are restarted from each turning point $\theta_p$, $\theta_a$ (roots of $E = U_{\text{eff}}(\theta)$) with the opposite sign.
}
\end{problem}

% ============ Problem 2 ============ %

\begin{problem}
{
Integrate the equations of motion for a particle moving on the surface of a cone (of vertical angle $2\alpha$) placed vertically and with vertex downwards in a gravitational field.
}
{
We proceed as in \probref{\S 14}{1}. By using the  spherical co-ordinate, we can write the position of the particle as
\begin{align*}
    x &= r\cos{\phi}\sin{\alpha} \\
    y &= r\sin{\phi}\sin{\alpha} \\
    z &= r\cos{\alpha}.
\end{align*}
By taking the time derivative of those, we obtain
\begin{align*}
    \Dot{x} &= \Dot{r}\cos{\phi}\sin{\alpha} - r\Dot{\phi}\sin{\phi}\sin{\alpha} \\
    \Dot{y} &= \Dot{r}\sin{\phi}\sin{\alpha} + r\Dot{\phi}\cos{\phi}\sin{\alpha} \\
    \Dot{z} &= \Dot{r}\cos{\alpha}.
\end{align*}
The Lagrangian is thus
\begin{align*}
    L &= \frac{1}{2}m\left( \Dot{x}^2 + \Dot{y}^2 + \Dot{z}^2 \right) - mgz \\
    &= \frac{1}{2}m\left( \Dot{r}^2 + r^2\Dot{\phi}^2\sin^2{\alpha} \right) -mgr\cos{\alpha} .
\end{align*}
The co-ordinate $\phi$ is cyclic, thus 
\begin{equation} \label{S14P2_Mz}
    M_z = p_{\phi} = \pd{L}{\Dot{\phi}} = mr^2\Dot{\phi}\sin^2{\alpha}
\end{equation}
is conserved. This is the $z$-component of the angular momentum found in \probref{\S 9}{1}, with the distance $r\sin{\alpha}$ from the axis. The energy of the particle is
\begin{equation} \label{S14P2_E}
    E = \frac{1}{2}m\left( \Dot{r}^2 + r^2\Dot{\phi}^2\sin^2{\alpha} \right) + mgr\cos{\alpha} .
\end{equation}
We can rewrite \eqref{S14P2_Mz} as
\begin{equation*}
    \Dot{\phi}^2 =  \frac{M_z^2}{m^2r^4\sin^4{\alpha}}
\end{equation*}
and substituting in \eqref{S14P2_E}, we get
\begin{align*}
    E &= \frac{1}{2}m\Dot{r}^2 + \frac{M_z^2}{2mr^2\sin^2{\alpha}} + mgr\cos{\alpha} \\
    &= \frac{1}{2}m\Dot{r}^2 + U_{\text{eff}}(r).
\end{align*}
Hence, on a branch where $r$ increases,
\begin{align*}
    t = \sqrt{\frac{m}{2}} \int \frac{dr}{\sqrt{E-U_{\text{eff}}(r)}}.
\end{align*}
From \eqref{S14P2_Mz}, it is possible to write, on the same branch,
\begin{align*}
    &\frac{d\phi}{dt} = \frac{M_z}{mr^2\sin^2{\alpha}} \\
    \Rightarrow &\frac{d\phi}{dr}\frac{dr}{dt} = \frac{M_z}{mr^2\sin^2{\alpha}} \\
    \Rightarrow &\frac{d\phi}{dr} \sqrt{\frac{2\left(E-U_{\text{eff}}(r)\right)}{m}} = \frac{M_z}{mr^2\sin^2{\alpha}} \\
    \Rightarrow &\phi =\frac{M_z}{\sqrt{2m}\sin^2{\alpha}} \int \frac{dr}{r^2\sqrt{E-U_{\text{eff}}(r)}}.
\end{align*}
Written with explicit limits, the co-ordinates evolve as
\begin{align*}
    t - t_0 &= \sqrt{\frac{m}{2}} \int_{r_0}^{r} \frac{ds}{\sqrt{E-U_{\text{eff}}(s)}}, \\
    \phi - \phi_0 &=\frac{M_z}{\sqrt{2m}\sin^2{\alpha}} \int_{r_0}^{r} \frac{ds}{s^2\sqrt{E-U_{\text{eff}}(s)}},
\end{align*}
where $r_0 = r(t_0)$, $\phi_0 = \phi(t_0)$ and $s$ is the integration variable. On a branch where $r$ decreases, $\Dot{r}$ is the negative square root and both right-hand sides change sign, so that $t$ always increases and $\phi$ always changes in the direction given by the sign of $M_z$. At a turning point, $\Dot{r}$ changes sign and the integrals are restarted from it with the opposite sign. It is possible to find the range of $r$. At the maximum and minimum value of $r$, the radial velocity is zero (the azimuthal velocity is not), thus
\begin{align*}
    E &= U_{\text{eff}} \\
    &= \frac{M_z^2}{2mr^2\sin^2{\alpha}} + mgr\cos{\alpha} \\
\end{align*}
We can find a cubic algebraic equation for $r$, that is
\begin{align*}
    0 &= r^2\left( E - mgr\cos{\alpha} \right) - \frac{M_z^2}{2m\sin^2{\alpha}} \\
    &= mgr^3\cos{\alpha} - Er^2 + \frac{M_z^2}{2m\sin^2{\alpha}}.
\end{align*}
When $M_z \neq 0$ and $E$ is above the minimum of $U_{\text{eff}}(r)$, this equation has two distinct positive roots, $r_p < r_a$, which are the turning points of the motion (the third root is negative). When $E$ equals the minimum of $U_{\text{eff}}$, the two roots merge and the orbit is a circle, $r = \text{constant}$. When $M_z = 0$, there is no centrifugal barrier at the vertex: the particle moves along a generator of the cone and reaches the vertex $r=0$. To summarize, the solution of the equation of motion for a particle moving on the surface of a cone is
}
{
\begin{align*}
  t - t_0 &=  \sqrt{\frac{m}{2}} \int_{r_0}^{r} \frac{ds}{\sqrt{E-U_{\text{eff}}(s)}} \\
  \phi - \phi_0 &= \frac{M_z}{\sqrt{2m}\sin^2{\alpha}} \int_{r_0}^{r} \frac{ds}{s^2\sqrt{E-U_{\text{eff}}(s)}} \\
  U_{\text{eff}}(r) &= \frac{M_z^2}{2mr^2\sin^2{\alpha}} + mgr\cos{\alpha}
\end{align*}
on a branch where $r$ increases; both right-hand sides change sign on a branch where $r$ decreases, and the integrals are restarted from each turning point $r_p$, $r_a$ (roots of $E = U_{\text{eff}}(r)$) with the opposite sign.
}
\end{problem}

% ============ Problem 3 ============ %

\begin{problem}
{
Integrate the equations of motion for a pendulum of mass $m_2$, with a mass $m_1$ at the point of support which can move on a horizontal line lying in the plane in which $m_2$ moves (see the figure in \probref{\S 5}{2}).
}
{
From \probref{\S 5}{2}, the Lagrangian of the system is
\begin{equation*}
    L = \frac{1}{2} (m_1 + m_2) \Dot{x}^2 +  \frac{1}{2} m_2 (l^2 \Dot{\phi}^2 + 2 l \Dot{x} \Dot{\phi} \cos{\phi} ) + m_2 g l \cos{\phi}.
\end{equation*}
The co-ordinate $x$ is cyclic, thus
\begin{equation} \label{S14P3_px}
    p_x = \pd{L}{\Dot{x}} = (m_1 + m_2) \Dot{x} + m_2 l \Dot{\phi} \cos{\phi} = \text{constant}
\end{equation}
is conserved. It is always possible to find an inertial frame of reference where $p_x=0$, using this frame we get
\begin{align}
    &(m_1 + m_2) \Dot{x} + m_2 l \Dot{\phi} \cos{\phi} = 0 \nonumber \\
    \Rightarrow &\int \left( (m_1 + m_2) \Dot{x} + m_2 l \Dot{\phi} \cos{\phi} \right)dt = \text{constant} \nonumber \\
    \Rightarrow &(m_1 + m_2) x + m_2 l \sin{\phi} = (m_1+m_2)R_x = \text{constant} . \label{S14P3_Rx}
\end{align}
This express the fact that the centre of mass $\mathbf{R}$ of the system does not move horizontally. It is also possible to rewrite \eqref{S14P3_px} as
\begin{align*}
    &(m_1 + m_2) \Dot{x} + m_2 l \Dot{\phi} \cos{\phi} = 0\\
    \Rightarrow&\Dot{x} = \frac{-m_2l\Dot{\phi}\cos{\phi}}{m_1+m_2}
\end{align*}
Plugging this into the energy of the system we get
\begin{align*}
    E &= \frac{1}{2} (m_1 + m_2) \Dot{x}^2 +  \frac{1}{2} m_2 (l^2 \Dot{\phi}^2 + 2 l \Dot{x} \Dot{\phi} \cos{\phi} ) - m_2 g l \cos{\phi} \\
    &= \frac{m_2^2l^2\Dot{\phi}^2\cos^2{\phi}}{2(m_1+m_2)} +  \frac{1}{2} m_2 l^2 \Dot{\phi}^2 -  \frac{m_2^2l^2\Dot{\phi}^2\cos^2{\phi}}{m_1+m_2} - m_2 g l \cos{\phi} \\
    &= \frac{1}{2}m_2l^2\Dot{\phi}^2 \left( 1 -\frac{m_2}{m_1+m_2}\cos^2{\phi} \right) - m_2 g l \cos{\phi}.
\end{align*}
Hence,
\begin{align*}
    t &= l\sqrt{\frac{m_2}{2}}\int\sqrt{\frac{1-\frac{m_2}{m_1+m_2}\cos^2{\phi}}{E+m_2gl\cos{\phi}}}d\phi \\
    &= l\sqrt{\frac{m_2}{2(m_1+m_2)}}\int\sqrt{\frac{m_1+m_2\sin^2{\phi}}{E+m_2gl\cos{\phi}}}d\phi.
\end{align*}
Using \eqref{S14P3_Rx}, we can express the position of the mass $m_2$ as
\begin{equation*}
    \begin{bmatrix} x_2 \\ y_2 \end{bmatrix} = \begin{bmatrix} x + l\sin{\phi} \\ l\cos{\phi} \end{bmatrix}=\begin{bmatrix} R_x + l\sin{\phi}\left(1 - \frac{m_2}{m_1+m_2} \right) \\ l\cos{\phi} \end{bmatrix} = \begin{bmatrix} R_x + \frac{m_1}{m_1+m_2}l\sin{\phi} \\ l\cos{\phi} \end{bmatrix},
\end{equation*}
since \eqref{S14P3_Rx} gives $x = R_x - \frac{m_2}{m_1+m_2}l\sin{\phi}$.
The path of the particle of mass $m_2$ is thus an arc of an ellipse center at $(R_x,0)$ with horizontal semi-axis $lm_1/(m_1+m_2)$ and vertical semi-axis $l$. It is possible to see that when $m_1 \rightarrow \infty$, the path return to the simple pendulum. The solution of the equations of motion is thus
}
{
\begin{align*}
  t &= l\sqrt{\frac{m_2}{2}}\int\sqrt{\frac{1-\frac{m_2}{m_1+m_2}\cos^2{\phi}}{E+m_2gl\cos{\phi}}}d\phi \\
  \begin{bmatrix} x_2 \\ y_2 \end{bmatrix} &= \begin{bmatrix} R_x + l\sin{\phi}\left(1 - \frac{m_2}{m_1+m_2} \right) \\ l\cos{\phi} \end{bmatrix} = \begin{bmatrix} R_x + \frac{m_1}{m_1+m_2}l\sin{\phi} \\ l\cos{\phi} \end{bmatrix}
\end{align*}
}
\end{problem}

% ============ Problem 1 ============ %

\begin{problem}
{
Find the time dependence of the co-ordinates of a particle with energy $E=0$ moving in a parabola in a field $U=-\alpha/r$.
}
{
From formula (14.6) of the book, we have the integral
\begin{align*}
    t &= \int \frac{dr}{\sqrt{\frac{2}{m}\left(E-U(r)\right)-\frac{M^2}{m^2r^2}}} \\
    &= \int \frac{dr}{\sqrt{\frac{2\alpha}{mr}-\frac{M^2}{m^2r^2}}} \\
    &= \int \frac{rdr}{\sqrt{\frac{2\alpha}{m}r-\frac{M^2}{m^2}}},
\end{align*}
written here for the outgoing branch ($\Dot{r}>0$), and taking $M>0$. We define the parameter of the parabola
\begin{equation*}
    p = \frac{M^2}{m\alpha}
\end{equation*}
and use the substitution
\begin{align}
    & r = \frac{M^2}{2m\alpha}\left(1+\eta^2\right) = \frac{1}{2}p\left(1+\eta^2\right), \label{S15P1_r} \\
    \Rightarrow & \sqrt{\frac{2\alpha}{m}r - \frac{M^2}{m^2}} = \sqrt{\frac{M^2}{m^2}\eta^2} = \frac{M}{m}|\eta| , \nonumber
\end{align}
with the differential form
\begin{equation*}
    dr = \frac{M^2}{m\alpha}\eta d\eta = p\,\eta\,d\eta.
\end{equation*}
On the outgoing branch we take $\eta \geq 0$, so that $|\eta| = \eta$, and the integral become
\begin{align*}
    t - t_0 &= \frac{m}{M}\int_0^\eta \frac{1}{2}p\left(1+\eta'^2\right) p \, d\eta' \\
    &= \frac{M^3}{2m\alpha^2}\int_0^\eta\left(1+\eta'^2\right)d\eta' \\
    &= \frac{M^3}{2m\alpha^2}\left(\eta+\frac{1}{3}\eta^3\right) \\
    &= \frac{1}{2}\sqrt{\frac{mp^3}{\alpha}}\left(\eta+\frac{1}{3}\eta^3\right),
\end{align*}
where $t_0$ is the time of passage at the perihelion $r = p/2$ ($\eta = 0$). On the incoming branch ($\Dot{r}<0$), the integrand of $t$ changes sign; taking there $\eta \leq 0$, so that $|\eta| = -\eta$, the two sign changes cancel and the same formula holds. It is thus important to specify that the parameter $\eta$ is signed and varies from $-\infty$ to $\infty$: $\eta<0$ before the perihelion and $\eta>0$ after it. Using \eqref{S15P1_r} and the equation of the parabola (eccentricity $e=1$)
\begin{equation*}
    \cos{\phi} = \frac{p}{r} - 1 = \frac{1-\eta^2}{1+\eta^2},
\end{equation*}
it is possible to find the Cartesian co-ordinates
\begin{equation*}
    x = r\cos{\phi} = \frac{1}{2}p\left(1-\eta^2\right)
\end{equation*}
and
\begin{equation*}
    y^2 = r^2 - x^2 = \frac{p^2}{4}\left[\left(1+\eta^2\right)^2-\left(1-\eta^2\right)^2\right] = p^2\eta^2 ,
\end{equation*}
whose branch is selected by the sign of $\eta$: for $M>0$ the particle moves counterclockwise, so $y = p\eta$ (negative before the perihelion, positive after it). The parametric form of the required dependence are thus
}
{
\begin{align*}
    r &= \frac{1}{2}p\left(1+\eta^2\right) & t - t_0 &= \frac{1}{2}\sqrt{\frac{mp^3}{\alpha}}\left(\eta+\frac{1}{3}\eta^3\right) \\
    x &= \frac{1}{2}p\left(1-\eta^2\right) & y &= p\eta
\end{align*}
with $p = M^2/(m\alpha)$ ($M>0$), the signed parameter $-\infty<\eta<\infty$, and $t_0$ the time of passage at the perihelion.
}
\end{problem}

% ============ Problem 2 ============ %

\begin{problem}
{
Integrate the equations of motion for a particle in a central field
\begin{align*}
    U &= -\frac{\alpha}{r^2} & &(\alpha > 0).
\end{align*}
}
{
The energy of the particle, written with the effective potential energy, is
\begin{equation*}
    E = \frac{1}{2}m\Dot{r}^2 + \frac{M^2}{2mr^2} - \frac{\alpha}{r^2} = \frac{1}{2}m\Dot{r}^2 - \frac{C}{2mr^2},
    \qquad C \equiv 2m\alpha - M^2 .
\end{equation*}
The real constant $C$ compares the attraction with the centrifugal barrier: the effective potential energy $-C/(2mr^2)$ is repulsive for $C<0$, attractive for $C>0$, and vanishes identically for the critical angular momentum $M^2 = 2m\alpha$ ($C=0$). From formula (14.6) of the book, we have
\begin{align*}
    dt &= \pm\frac{dr}{\sqrt{\frac{2}{m}\left(E-U(r)\right)-\frac{M^2}{m^2r^2}}} \\
    &= \pm\frac{dr}{\sqrt{\frac{2E}{m}+\frac{C}{m^2r^2}}} \\
    &= \pm\frac{mr\,dr}{\sqrt{2mEr^2+C}},
\end{align*}
with the upper sign on a branch where $r$ increases and the lower sign on a branch where $r$ decreases.

\vspace{2ex}\textbf{Time dependence of $r$.} Rather than integrating this case by case, which would require dividing by $E$, we use the radial equation of motion, obtained by differentiating the energy with respect to time,
\begin{equation*}
    m\Ddot{r} = -\frac{C}{mr^3}.
\end{equation*}
With the energy equation $\Dot{r}^2 = 2E/m + C/(m^2r^2)$, it gives
\begin{equation*}
    \frac{d^2}{dt^2}\left(r^2\right) = 2\Dot{r}^2 + 2r\Ddot{r} = 2\left(\frac{2E}{m}+\frac{C}{m^2r^2}\right) - \frac{2C}{m^2r^2} = \frac{4E}{m},
\end{equation*}
a constant. Integrating twice from a reference time $t_{\text{ref}}$, at which $r = r_0$ and $\Dot{r} = \Dot{r}_0$, we get
\begin{equation} \label{S15P2_r2}
    r^2(t) = r_0^2 + 2r_0\Dot{r}_0\left(t-t_{\text{ref}}\right) + \frac{2E}{m}\left(t-t_{\text{ref}}\right)^2,
\end{equation}
where the initial data must satisfy the energy equation, $\Dot{r}_0^2 = 2E/m + C/(m^2r_0^2)$. This law is valid on each physical segment of the motion, \ie as long as $r>0$: if $r^2$ reaches zero, the particle falls into the centre, where the field is singular. It covers the case $E=0$, where $r^2$ is a linear function of time, without any division by zero. When $E \neq 0$, completing the square and using the energy equation give
\begin{equation*}
    r^2 = \frac{2E}{m}\left(t-t_c\right)^2 - \frac{C}{2mE},
    \qquad t_c = t_{\text{ref}} - \frac{mr_0\Dot{r}_0}{2E},
\end{equation*}
which is also what the direct integration of $dt$ gives (with the substitution $w = 2mEr^2 + C$, $dw = 4mEr\,dr$):
\begin{equation*}
    t - t_c = \pm\frac{1}{2E}\sqrt{2mEr^2+C} .
\end{equation*}

\vspace{2ex}\textbf{Equation of the path.} Formula (14.7) of the book gives, on the same branches,
\begin{align*}
    d\phi &= \pm\frac{M\,dr}{r^2\sqrt{2m\left(E-U(r)\right)-\frac{M^2}{r^2}}} \\
    &= \pm\frac{M\,dr}{r^2\sqrt{2mE+\frac{C}{r^2}}}.
\end{align*}
Using the substitution
\begin{equation*}
    u = \frac{1}{r}
\end{equation*}
with the differential form
\begin{equation*}
    du = -\frac{1}{r^2}dr,
\end{equation*}
it becomes
\begin{equation} \label{S15P2_phi}
    d\phi = -\frac{M\,du}{\sqrt{2mE + Cu^2}}
\end{equation}
on a branch where $r$ increases ($u$ decreases), with the opposite sign on a branch where $r$ decreases. In both cases $\phi$ changes in the direction of $\Dot{\phi} = M/(mr^2)$. Rather than integrating \eqref{S15P2_phi} with a different substitution for each sign of $C$ and $E$, we take $M>0$ (the case $M<0$ follows by changing $\phi$ into $-\phi$) and consider $u$ as a function of $\phi$, primes meaning differentiation with respect to $\phi$. Since $\Dot{r} = -u'\Dot{\phi}/u^2 = -Mu'/m$, the energy becomes
\begin{equation*}
    E = \frac{M^2}{2m}\left(u'^2+u^2\right) - \alpha u^2,
\end{equation*}
\ie
\begin{equation} \label{S15P2_first}
    (u')^2 - \frac{C}{M^2}u^2 = \frac{2mE}{M^2},
\end{equation}
which is also \eqref{S15P2_phi} squared. Differentiating with respect to $\phi$ (or using the Binet equation $u''+u = -\frac{m}{M^2}\frac{dU}{du}$ with $U = -\alpha u^2$) gives the linear equation
\begin{equation*}
    u'' - \frac{C}{M^2}u = 0 .
\end{equation*}
Its real solutions must also satisfy the first integral \eqref{S15P2_first}, which fixes their amplitude, and they describe a motion only where $u>0$. This gives the following orbit families.
\begin{itemize}
    \item $C<0$, $E>0$: $u = \sqrt{\frac{2mE}{-C}}\cos{\left(\frac{\sqrt{-C}}{M}(\phi-\phi_0)\right)}$, with $|\phi-\phi_0| < \frac{\pi M}{2\sqrt{-C}}$. The particle comes from infinity, reaches the minimum distance $r_{\text{min}} = \sqrt{-C/(2mE)}$ at $\phi = \phi_0$ and goes back to infinity, after turning through the angle $\pi M/\sqrt{-C} > \pi$.
    \item $C>0$, $E>0$: $u = \sqrt{\frac{2mE}{C}}\sinh{\left(\frac{\sqrt{C}}{M}(\phi-\phi_0)\right)}$, with $\phi>\phi_0$. The particle comes from infinity ($\phi\rightarrow\phi_0$) and spirals into the centre ($\phi\rightarrow\infty$). The reflected branch, $u = \sqrt{\frac{2mE}{C}}\sinh{\left(\frac{\sqrt{C}}{M}(\phi_0-\phi)\right)}$ with $\phi<\phi_0$, describes a particle spiralling out of the centre to infinity.
    \item $C>0$, $E<0$: $u = \sqrt{\frac{2m|E|}{C}}\cosh{\left(\frac{\sqrt{C}}{M}(\phi-\phi_0)\right)}$. The particle spirals out of the centre, reaches the maximum distance $r_{\text{max}} = \sqrt{C/(2m|E|)}$ at $\phi = \phi_0$, and spirals back into the centre.
    \item $C>0$, $E=0$: $u = A\exp{\left[\pm\frac{\sqrt{C}}{M}(\phi-\phi_0)\right]}$, $A>0$, a logarithmic spiral, traversed inwards (upper sign) or outwards (lower sign).
    \item $C=0$, $E>0$: $u = \pm\frac{\sqrt{2mE}}{M}(\phi-\phi_0)$, with the sign making $u>0$: a hyperbolic spiral, traversed inwards for $\phi>\phi_0$ (upper sign) or outwards for $\phi<\phi_0$ (lower sign).
    \item $C=0$, $E=0$: $u = A>0$, a uniform circular motion of arbitrary positive radius $1/A$, the centrifugal barrier exactly balancing the attraction.
\end{itemize}
One verifies each of them by substitution in the linear equation and in \eqref{S15P2_first}; for instance, for $C<0$, $u'^2 + \frac{|C|}{M^2}u^2 = \frac{2mE}{-C}\frac{|C|}{M^2} = \frac{2mE}{M^2}$. The time dependence along each path is given by \eqref{S15P2_r2}. For $E\neq0$, since $\frac{d}{dt}\left(r^2\right) = 2r\Dot{r} = -\frac{2M}{m}\frac{u'}{u}$ and $\frac{d}{dt}\left(r^2\right) = \frac{4E}{m}(t-t_c)$, we have the equivalent parameter relation
\begin{equation*}
    t - t_c = -\frac{M}{2E}\frac{u'}{u},
\end{equation*}
which fixes the orientation consistently with $\Dot{\phi} = M/(mr^2) > 0$. For example,
\begin{align*}
    t - t_c &= \frac{\sqrt{-C}}{2E}\tan{\left(\frac{\sqrt{-C}}{M}(\phi-\phi_0)\right)} & &(C<0,\ E>0), \\
    t - t_c &= \frac{\sqrt{C}}{2|E|}\tanh{\left(\frac{\sqrt{C}}{M}(\phi-\phi_0)\right)} & &(C>0,\ E<0),
\end{align*}
both increasing with $\phi$. For $C>0$ (and for $C=0$, $E>0$), the particle falls into (or emerges from) the centre at a finite time, while turning an infinite number of times around it.

\vspace{2ex}\textbf{Radial motion, $M=0$.} Then $C = 2m\alpha>0$, the direction $\phi$ is constant and the particle moves on a straight line through the centre. Its distance follows the same law \eqref{S15P2_r2}, for any energy, until it reaches the singular centre.

\vspace{2ex}\textbf{No motion.} Since $\Dot{r}^2 = 2E/m + C/(m^2r^2)$ cannot be negative, there is no motion for $C<0$ and $E\leq0$, nor for $C=0$ and $E<0$.

It is also interesting to know that these paths are Cotes's spirals\footnote{Whittaker ET (1937). A Treatise on the Analytical Dynamics of Particles and Rigid Bodies, with an Introduction to the Problem of Three Bodies (4th ed.). New York: Dover Publications. pp. 80–83. ISBN 978-0-521-35883-5.}. To summarize, the equations of motion for a particle in a central inverse-cube law force are
}
{
With $C = 2m\alpha - M^2$, $u = 1/r$ and $M>0$ (change $\phi$ into $-\phi$ for $M<0$), the paths are, where $u>0$,
\begin{align*}
    C<0,\ E>0: &\quad u = \sqrt{\frac{2mE}{-C}}\cos{\left(\frac{\sqrt{-C}}{M}(\phi-\phi_0)\right)} \\
    C>0,\ E>0: &\quad u = \sqrt{\frac{2mE}{C}}\sinh{\left(\frac{\sqrt{C}}{M}(\phi-\phi_0)\right)} \quad \text{(or its reflection } \phi-\phi_0 \rightarrow \phi_0-\phi\text{)} \\
    C>0,\ E<0: &\quad u = \sqrt{\frac{2m|E|}{C}}\cosh{\left(\frac{\sqrt{C}}{M}(\phi-\phi_0)\right)} \\
    C>0,\ E=0: &\quad u = A\exp{\left[\pm\frac{\sqrt{C}}{M}(\phi-\phi_0)\right]}, \qquad A>0 \\
    C=0,\ E>0: &\quad u = \pm\frac{\sqrt{2mE}}{M}(\phi-\phi_0) \\
    C=0,\ E=0: &\quad u = A>0 \quad \text{(circular motion of arbitrary radius)}
\end{align*}
On each segment of the motion (until $r$ reaches the centre),
\begin{align*}
    r^2(t) &= r_0^2 + 2r_0\Dot{r}_0(t-t_{\text{ref}}) + \frac{2E}{m}(t-t_{\text{ref}})^2, &
    \Dot{r}_0^2 &= \frac{2E}{m} + \frac{C}{m^2r_0^2},
\end{align*}
and, for $E\neq0$, $t - t_c = -\frac{M}{2E}\frac{u'}{u}$. For $M=0$, the motion is radial, with constant $\phi$ and the same $r^2(t)$. There is no motion for $C<0$, $E\leq0$, nor for $C=0$, $E<0$.
}
\end{problem}

% ============ Problem 3 ============ %

\begin{problem}
{
When a small correction $\delta U(r)$ is added to the potential energy $U=-\alpha/r$, the paths of finite motion are no longer closed, and at each revolution the perihelion is displaced through a small angle $\delta\phi$. Find $\delta\phi$ when
}{}{}
\end{problem}
\begin{subproblem}
{
\begin{equation*}
    \delta U = \frac{\beta}{r^2}
\end{equation*}
}
{
From formula (14.10) of the book, we have
\begin{align*}
    \Delta \phi &= 2\int_{r_{\text{min}}}^{r_{\text{max}}}\frac{Mdr}{r^2\sqrt{2m(E-U)-\frac{M^2}{r^2}}} \\
    &= -2\pd{}{M}\int_{r_{\text{min}}}^{r_{\text{max}}}\sqrt{2m(E-U)-\frac{M^2}{r^2}}dr \\
    &= -2\pd{}{M}\int_{r_{\text{min}}}^{r_{\text{max}}}\sqrt{2m\left(E+\frac{\alpha}{r}-\delta U\right)-\frac{M^2}{r^2}}dr.
\end{align*}
Expanding the integrand in powers of $\delta U$ involves using a Taylor series expansion. Let's denote the integrand as $F$:
\begin{equation*}
    F = \sqrt{2m\left(E+\frac{\alpha}{r}-\delta U\right)-\frac{M^2}{r^2}} = \sqrt{F_0^2 - 2m\,\delta U},
\end{equation*}
where
\begin{equation*}
    F_0 = \sqrt{2m\left(E+\frac{\alpha}{r}\right)-\frac{M^2}{r^2}}
\end{equation*}
is the value of $F$ for the unperturbed motion. We want to expand $F$ around $\delta U = 0$. The expansion will look like:
\begin{equation*}
    F = F_0 + F_1 \delta U + F_2 (\delta U)^2 + \ldots
\end{equation*}
Here, the Taylor coefficients are $F_k = \frac{1}{k!}\frac{\partial^k F}{\partial (\delta U)^k}\big|_{\delta U=0}$: $F_0$ is the value of $F$ at $\delta U = 0$, $F_1$ is the first derivative with respect to $\delta U$ at $\delta U = 0$, $F_2$ is half the second derivative, and so on. Let's find the derivatives:
\begin{align*}
    \frac{\partial F}{\partial (\delta U)} &= -\frac{m}{\sqrt{F_0^2 - 2m\,\delta U}}, &
    \frac{\partial F}{\partial (\delta U)}\Big|_{\delta U=0} &= -\frac{m}{F_0}, \\
    \frac{\partial^2 F}{\partial (\delta U)^2} &= -\frac{m^2}{\left(F_0^2 - 2m\,\delta U\right)^{3/2}}, &
    \frac{\partial^2 F}{\partial (\delta U)^2}\Big|_{\delta U=0} &= -\frac{m^2}{F_0^3},
\end{align*}
so that $F_1 = -m/F_0$ and $F_2 = -m^2/(2F_0^3)$. Therefore, the expanded expression in powers of $\delta U$ is:
\begin{equation*}
    F = F_0 - \frac{m\,\delta U}{F_0} - \frac{m^2(\delta U)^2}{2F_0^3} + \ldots
\end{equation*}
After plugging the expended expression of $F$ in the integral, we see that the zero-order term gives $2\pi$. The first-order term gives the required change $\delta \phi$ :
\begin{equation} \label{S15P3_phi}
    \delta \phi = \pd{}{M}\int_{r_{\text{min}}}^{r_{\text{max}}} \frac{2m\,\delta U}{F_0}\,dr.
\end{equation}
We can change the integration over $r$ to one over $\phi$, along the path of the unperturbed motion. On its outward branch, from the perihelion ($r = r_{\text{min}}$, $\phi = 0$) to the aphelion ($r = r_{\text{max}}$, $\phi = \pi$), formula (14.7) of the book gives
\begin{equation*}
    d\phi = \frac{M\,dr}{r^2F_0},
    \qquad \text{\ie} \qquad
    \frac{dr}{F_0} = \frac{r^2}{M}\,d\phi .
\end{equation*}
Hence, the integral \eqref{S15P3_phi} become
\begin{align}
    \delta \phi &= \pd{}{M}\left( \frac{2m}{M} \int_0^\pi r^2\delta Ud\phi \right) \label{S15P3_dphi} \\
    &= \pd{}{M}\left( \frac{2m}{M} \int_0^\pi r^2\frac{\beta}{r^2}d\phi \right) \nonumber \\
    &= \pd{}{M}\left( \frac{2\pi m\beta}{M} \right) \nonumber  \\
    &= -\frac{2\pi m\beta}{M^2}. \nonumber
\end{align}
Using formula (15.4) of the book we can finally write
}
{
\begin{equation*}
    \delta \phi = -\frac{2\pi\beta}{\alpha p}
\end{equation*}
}
\end{subproblem}
\begin{subproblem}
{
\begin{equation*}
    \delta U = \frac{\gamma}{r^3}
\end{equation*}
}
{
From \eqref{S15P3_dphi}, we have
\begin{equation*}
     \delta \phi = \pd{}{M}\left( \frac{2m}{M} \int_0^\pi \frac{\gamma}{r}d\phi \right).
\end{equation*}
Using formula (15.5) of the book, the integral become
\begin{align*}
    \delta \phi &= \pd{}{M}\left( \frac{2m}{M} \int_0^\pi \frac{\gamma}{p}\left(1+e\cos\phi\right)d\phi \right) \\
    &= \pd{}{M}\left( \frac{2\pi\gamma m}{Mp} \right) \\
    &= -\frac{6\pi\alpha\gamma m^2}{M^4}.
\end{align*}
The last expression can also be written as
}
{
\begin{equation*}
    \delta \phi = -\frac{6\pi\gamma}{\alpha p^2}
\end{equation*}
}
\end{subproblem}
